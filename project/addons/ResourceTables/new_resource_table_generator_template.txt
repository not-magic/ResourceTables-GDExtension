@tool
class_name ${RESOURCE_CLASS_NAME}Generator
extends ResourceTableGenerator

func _init() -> void:
	super("${RESOURCE_CLASS_NAME}")

func _generate_outputs() -> void:
	var output := ResourceTableUtils.find_or_create("GenericResourceTable", "${RESOURCE_TABLE_DIR}${RESOURCE_TABLE_NAME}.tres")
	output.items = ResourceTableUtils.find_resources_of_type("${RESOURCE_CLASS_NAME}")
	ResourceSaver.save(output)
